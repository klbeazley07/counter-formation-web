869	
870	export function RhythmPage() {
871	  const { rhythm }  = useParams();
872	  const navigate    = useNavigate();
873	  const rfillRef    = useRef(null);
874	  const data        = RHYTHMS.find(r => r.slug === rhythm);
875	
876	  useEffect(() => { if (!data) navigate("/", { replace: true }); }, [data, navigate]);
877	  useEffect(() => { window.scrollTo(0, 0); }, [rhythm]);
878	  useEffect(() => {
879	    if (!data) return;
880	    const onScroll = () => {
881	      const d = document.documentElement;
882	      const pct = d.scrollTop / (d.scrollHeight - d.clientHeight) || 0;
883	      if (rfillRef.current) rfillRef.current.style.width = (pct * 100) + "%";
884	    };
885	    window.addEventListener("scroll", onScroll, { passive: true });
886	    return () => window.removeEventListener("scroll", onScroll);
887	  }, [data]);
888	
889	  if (!data) return null;
890	
891	  const idx         = RHYTHMS.findIndex(r => r.slug === rhythm);
