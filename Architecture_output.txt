591	/* ─── ARCHITECTURE SLIDER ─────────────────────────────────────────── */
592	
593	// How many wheel "ticks" to consume before advancing a panel or releasing
594	const TICKS_TO_ADVANCE = 5;   // ticks to move to next panel
595	const TICKS_TO_RELEASE = 1;   // extra ticks on last panel before exiting
596	
597	export function ArchitectureSlider() {
598	  const outerRef    = useRef(null);
599	  const trackRef    = useRef(null);
600	  const touchStartRef = useRef(null);
601	  const [idx, setIdx] = useState(0);
602	  const TOTAL = PILLARS.length;
603	
604	  // Accumulated wheel delta — use a ref so wheel handler closure stays fresh
605	  const accumRef    = useRef(0);
606	  const lockedRef   = useRef(false); // debounce between advances
607	  const exitAccRef  = useRef(0);     // extra ticks on last panel before release
608	
609	  // ── Apply track translation whenever idx changes
610	  useEffect(() => {
611	    const track = trackRef.current;
612	    if (!track) return;
613	    track.style.transform = `translateX(-${idx * (100 / TOTAL)}%)`;
614	
615	    // Update is-active class
616	    track.querySelectorAll(".arch-panel").forEach((p, i) => {
617	      p.classList.toggle("is-active", i === idx);
618	    });
619	
620	    // Reset accumulators when panel changes
621	    accumRef.current = 0;
622	    exitAccRef.current = 0;
623	  }, [idx, TOTAL]);
624	
625	  // ── Mark first panel active on mount
626	  useEffect(() => {
627	    const track = trackRef.current;
628	    if (!track) return;
629	    const panels = track.querySelectorAll(".arch-panel");
630	    if (panels[0]) panels[0].classList.add("is-active");
631	  }, []);
632	
633	  // ── Snap into viewport — fires on scroll, only during manual wheel/touch
634	  useEffect(() => {
635	    const outer = outerRef.current;
636	    if (!outer || window.innerWidth < 768) return;
637	
638	    let snapping = false;
639	    let lastInputTime = 0; // timestamp of last wheel/touch — avoids timeout race
640	
641	    const markInput = () => { lastInputTime = Date.now(); };
642	
643	    const onScroll = () => {
644	      if (snapping) return;
645	      // Only act on genuine user input (wheel/touch within last 400ms)
646	      if (Date.now() - lastInputTime > 400) return;
647	      const rect = outer.getBoundingClientRect();
648	      const vh = window.innerHeight;
649	
650	      // Snap DOWN: section top entering view from below
651	      if (rect.top > 10 && rect.top < vh * 0.65) {
652	        snapping = true;
653	        window.scrollTo({ top: outer.offsetTop, behavior: "smooth" });
654	        setTimeout(() => { snapping = false; }, 1200);
655	        return;
656	      }
657	
658	      // Snap UP: scrolling up and near the top of section (barely passed it)
659	      if (rect.top < -10 && rect.top > -vh * 0.65) {
660	        snapping = true;
661	        window.scrollTo({ top: outer.offsetTop, behavior: "smooth" });
662	        setTimeout(() => { snapping = false; }, 1200);
663	      }
664	    };
665	
666	    window.addEventListener("wheel",     markInput, { passive: true });
667	    window.addEventListener("touchmove", markInput, { passive: true });
668	    window.addEventListener("scroll",    onScroll,  { passive: true });
669	
670	    return () => {
671	      window.removeEventListener("wheel",     markInput);
672	      window.removeEventListener("touchmove", markInput);
673	      window.removeEventListener("scroll",    onScroll);
674	    };
675	  }, []);
676	
677	  // ── Wheel handler — intercepts scroll when section is pinned
678	  useEffect(() => {
679	    const outer = outerRef.current;
680	    if (!outer) return;
681	
682	    const onWheel = (e) => {
683	      // Only intercept on desktop
684	      if (window.innerWidth < 768) return;
685	
686	      // Is the section currently sticky (i.e. top of viewport)?
687	      const rect = outer.getBoundingClientRect();
688	      const isStuck = rect.top <= 2 && rect.top >= -2;
689	      if (!isStuck) return;
690	
691	      // Prevent native scroll while we're intercepting
692	      e.preventDefault();
693	
694	      const dir = e.deltaY > 0 ? 1 : -1;
695	
696	      // Going backward — always allow immediately
697	      if (dir < 0) {
698	        if (idx > 0) {
699	          setIdx(i => i - 1);
700	        } else {
701	          // Already on first panel — jump to previous section
702	          const prev = outer.previousElementSibling;
703	          if (prev) window.scrollTo({ top: prev.offsetTop + prev.offsetHeight - window.innerHeight, behavior: "smooth" });
704	        }
705	        return;
706	      }
707	
708	      // Going forward
709	      if (lockedRef.current) return;
710	
711	      accumRef.current += 1;
712	
713	      // On last panel, require extra ticks before releasing
714	      if (idx === TOTAL - 1) {
715	        exitAccRef.current += 1;
716	        if (exitAccRef.current >= TICKS_TO_RELEASE) {
717	          exitAccRef.current = 0;
718	          accumRef.current = 0;
719	          const next = outer.nextElementSibling;
720	          const target = next ? next.offsetTop : outer.offsetTop + outer.offsetHeight;
721	          window.scrollTo({ top: target, behavior: "smooth" });
722	        }
723	        // Don't advance — already on last panel
724	        return;
725	      }
726	
727	      if (accumRef.current >= TICKS_TO_ADVANCE) {
728	        accumRef.current = 0;
729	        lockedRef.current = true;
730	        setIdx(i => Math.min(TOTAL - 1, i + 1));
731	        // Brief lock to prevent runaway advances on fast trackpads
732	        setTimeout(() => { lockedRef.current = false; }, 600);
733	      }
734	    };
735	
736	    // Must be non-passive to call preventDefault
737	    outer.addEventListener("wheel", onWheel, { passive: false });
738	    return () => outer.removeEventListener("wheel", onWheel);
739	  }, [idx, TOTAL]);
740	
741	  // ── Keyboard
742	  useEffect(() => {
743	    const onKey = (e) => {
744	      const outer = outerRef.current;
745	      if (!outer) return;
746	      const rect = outer.getBoundingClientRect();
747	      if (rect.top > 10 || rect.bottom < window.innerHeight - 10) return;
748	      if (e.key === "ArrowRight" || e.key === "ArrowDown") {
749	        e.preventDefault();
750	        if (idx < TOTAL - 1) setIdx(i => i + 1);
751	      }
752	      if (e.key === "ArrowLeft" || e.key === "ArrowUp") {
753	        e.preventDefault();
754	        if (idx > 0) setIdx(i => i - 1);
755	      }
756	    };
757	    window.addEventListener("keydown", onKey);
758	    return () => window.removeEventListener("keydown", onKey);
759	  }, [idx, TOTAL]);
760	
761	  // ── Touch/swipe handling for mobile
762	  useEffect(() => {
763	    const el = outerRef.current;
764	    if (!el) return;
765	
766	    const onTouchStart = (e) => {
767	      touchStartRef.current = { x: e.touches[0].clientX, y: e.touches[0].clientY };
768	    };
769	
770	    const onTouchEnd = (e) => {
771	      if (!touchStartRef.current) return;
772	      const dx = e.changedTouches[0].clientX - touchStartRef.current.x;
773	      const dy = e.changedTouches[0].clientY - touchStartRef.current.y;
774	      touchStartRef.current = null;
775	
776	      // Only register horizontal swipes (more horizontal than vertical, minimum 50px)
777	      if (Math.abs(dx) < 50 || Math.abs(dx) < Math.abs(dy)) return;
778	
779	      if (dx < 0) {
780	        // Swipe left = next
781	        setIdx(prev => Math.min(prev + 1, PILLARS.length - 1));
782	      } else {
783	        // Swipe right = previous
784	        setIdx(prev => Math.max(prev - 1, 0));
785	      }
786	    };
787	
788	    el.addEventListener('touchstart', onTouchStart, { passive: true });
789	    el.addEventListener('touchend', onTouchEnd, { passive: true });
790	    return () => {
791	      el.removeEventListener('touchstart', onTouchStart);
792	      el.removeEventListener('touchend', onTouchEnd);
793	    };
794	  }, []);
795	
796	  // ── Continue button (last panel)
797	  const handleContinue = useCallback(() => {
798	    const outer = outerRef.current;
799	    if (!outer) return;
800	    const next = outer.nextElementSibling;
801	    const target = next ? next.offsetTop : outer.offsetTop + outer.offsetHeight;
802	    window.scrollTo({ top: target, behavior: "smooth" });
803	  }, []);
804	
805	  const progPct = ((idx) / (TOTAL - 1)) * 100;
806	
807	  return (
808	    <section ref={outerRef} id="architecture" className="arch-outer">
809	      <div className="arch-sticky">
810	        <div className="arch-viewport">
811	
812	          {/* Counter */}
813	          <div className="arch-counter">
814	            <strong>{idx + 1}</strong> / {TOTAL}
815	          </div>
816	
817	          {/* Scroll-past hint */}
818	          <div className="arch-scroll-hint">
819	            <span>Scroll to explore · scroll past to skip</span>
820	            <svg width="12" height="12" viewBox="0 0 12 12" fill="none">
821	              <path d="M6 1v10M2 7l4 4 4-4" stroke="rgba(255,255,255,0.2)" strokeWidth="1.4" strokeLinecap="round"/>
822	            </svg>
823	          </div>
824	
825	          {/* Track */}
826	          <div ref={trackRef} className="arch-track">
827	            {PILLARS.map((p, i) => (
828	              <div key={p.slug} className={`arch-panel${i === 0 ? " is-active" : ""}`}>
829	                <div className="arch-bg" style={{ backgroundImage: `url('${p.img}')` }} />
830	                <div className="arch-ov-bottom" />
831	                <div className="arch-ov-top" />
832	                <div className="arch-content">
833	                  <div className="arch-eyebrow">
834	                    Pillar {p.num}&nbsp;&nbsp;·&nbsp;&nbsp;Architecture of the Soul
835	                  </div>
836	                  <h2 className="arch-headline">{p.challenge}</h2>
837	                  <p className="arch-manifesto">{p.manifesto}</p>
838	                  <div className="arch-cta-row">
839	                    <Link to={p.route} className="arch-cta-enter">
840	                      {p.cta}
841	                      <svg width="12" height="12" viewBox="0 0 12 12" fill="none">
842	                        <path d="M1 6h10M6.5 2l4 4-4 4" stroke="currentColor" strokeWidth="1.5" strokeLinecap="round"/>
843	                      </svg>
844	                    </Link>
845	                  </div>
846	                </div>
847	              </div>
848	            ))}
849	          </div>
850	
851	          {/* Right-side UI: dots + gold chevron */}
852	          <div className="arch-right-ui">
853	            {PILLARS.map((p, i) => (
854	              <button
855	                key={p.slug}
856	                className={`arch-rdot${i === idx ? " active" : ""}`}
857	                onClick={() => setIdx(i)}
858	                aria-label={`Go to ${p.title}`}
859	              />
860	            ))}
861	
862	            <div className="arch-right-sep" />
863	
864	            <button
865	              className="arch-right-next"
866	              onClick={() => idx < TOTAL - 1 ? setIdx(i => i + 1) : handleContinue()}
867	              disabled={false}
868	              aria-label="Next"
869	            >
870	              <svg width="18" height="18" viewBox="0 0 18 18" fill="none">
871	                <path d="M4 7l5 5 5-5" stroke="#C9A84C" strokeWidth="1.8" strokeLinecap="round" strokeLinejoin="round"/>
872	              </svg>
873	            </button>
874	          </div>
875	
876	          {/* Continue — last panel */}
877	          <button
878	            className={`arch-continue${idx === TOTAL - 1 ? " visible" : ""}`}
879	            onClick={handleContinue}
880	          >
881	            <span>Continue</span>
882	            <svg width="16" height="10" viewBox="0 0 16 10" fill="none">
883	              <path d="M1 1L8 8L15 1" stroke="rgba(255,255,255,0.35)" strokeWidth="1.5" strokeLinecap="round"/>
884	            </svg>
885	          </button>
886	
887	          {/* Progress bar */}
888	          <div className="arch-prog">
889	            <div className="arch-prog-fill" style={{ width: `${progPct}%` }} />
890	          </div>
891	
892	        </div>
893	      </div>
894	    </section>
895	  );
896	}
